# Pinned build inputs. Both workflows install from this file, so the version
# that goes into a released MSI is a decision recorded here rather than
# whatever PyPI served that morning.
#
# pysnmp is the one that matters most. The agent pre-computes BER on the wire
# and the bytes have to match what pyasn1 actually emits, which is not always
# the shortest form. A minor upgrade that changed an encoded length would push
# responses past the 1400-byte cap, and the symptom at the LibreNMS end is
# intermittent missing data, not an error. tests/test_ber_size.py checks the
# encoder against real output, but only for the version installed when it runs.
#
# These are the versions the verified builds were made with, read off the
# machine that made them.
pysnmp==7.1.29
pyasn1==0.6.4

# Pulled in by pysnmp, not imported by us directly, and it is already inside
# the 1.0.0 MSI: importing pysnmp.entity.config reaches the DES and AES
# modules, which is where this comes from. It was left unpinned while v2c
# was the only thing shipping, because nothing on the serving path touched
# it. SNMPv3 turns it into a load-bearing dependency -- every authPriv
# packet runs through its AES -- so a version that changes behaviour would
# be a functional outage, not dead weight. Pinned for the same reason as
# everything else in this file.
#
# There is a second, sharper reason to pin it. SNMPv3 privacy is AES in CFB
# mode, and cryptography is moving CFB out of primitives.ciphers.modes into
# decrepit, warning on every use. pysnmp 7.1.29 still imports the deprecated
# path. The day that removal completes, every authPriv packet stops working --
# not degrades, stops -- and the fix would have to come from pysnmp. Raising
# this pin is therefore not routine: run tests/test_usm_wire.py against a real
# net-snmp before believing an upgrade is safe.
cryptography==50.0.0

# Windows-only. Listed with a marker so the same file works on the Linux test

pywin32==312 ; sys_platform == "win32"
pyinstaller==6.22.2 ; sys_platform == "win32"
pyinstaller-hooks-contrib==2026.6 ; sys_platform == "win32"

pytest==9.1.1
